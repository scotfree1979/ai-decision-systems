v_oc_timing|CREATE VIEW v_oc_timing AS

      ((julianday(oc.snapshot_ts) - julianday(rt.off_time_utc)) * 24 * 60 * 60)

      WHEN ((julianday(oc.snapshot_ts) - julianday(rt.off_time_utc)) * 24 * 60)
           > (rt.expected_seconds / 60) THEN 'POST'
      ELSE 'UNKNOWN'
    END AS phase

  ON oc.marketId = rt.marketId
WHERE oc.snapshot_ts IS NOT NULL
  AND rt.expected_seconds IS NOT NULL
  AND (julianday(rt.off_time_utc) - julianday(oc.snapshot_ts)) * 24 * 60 BETWEEN -240 AND 30
GROUP BY oc.marketId, oc.stage, oc.selectionId

              WHEN UPPER(SUBSTR(COALESCE(source,''),1,1))='E' THEN 'IP4_FENCE_ERROR'
              WHEN UPPER(SUBSTR(COALESCE(source,''),1,1))='K' THEN 'IP5_COLLAPSE_FADE'
              WHEN UPPER(SUBSTR(COALESCE(source,''),1,1))='S' THEN 'LEGACY_S'

          FROM orders

        ),
        mode_sel AS (SELECT * FROM base WHERE UPPER(mode)=UPPER('TEST') ),
        open_parents AS (
          SELECT id FROM mode_sel

          FROM mode_sel

          SUM((bl.role IS NULL OR bl.role='PARENT') AND bl.es='PLACED')                                  AS p_p,

          SUM(UPPER(COALESCE(bl.role,''))='CHILD' AND bl.es='PLACED')                                    AS c_p,
          SUM(UPPER(COALESCE(bl.role,''))='CHILD' AND (bl.es='MATCHED' OR bl.xs='MATCHED'))              AS c_m,
          (SELECT COUNT(*) FROM open_parents op JOIN mode_sel b2 ON b2.id=op.id AND b2.letter=bl.letter) AS open_parents,
          ROUND(COALESCE((SELECT v FROM liab_matched_open WHERE letter=bl.letter),0),2)                  AS matched_liab,
          ROUND(COALESCE((SELECT v FROM liab_unmatched   WHERE letter=bl.letter),0),2)                    AS unmatched_liab,
          ROUND(COALESCE((SELECT pnl FROM pnl_today      WHERE letter=bl.letter),0),2)                    AS pnl_today,
          COALESCE((SELECT last_ts FROM last_trade       WHERE letter=bl.letter),'')                      AS last_trade_ts
        FROM mode_sel bl
        GROUP BY bl.letter, bl.strategy_name
        ORDER BY strategy

volatility_meta|CREATE VIEW volatility_meta AS
WITH prev AS (
  SELECT
      a.marketId,
      a.selectionId,
      a.id AS id_a,
      MAX(b.id) AS id_prev
  FROM oc_series a
  JOIN oc_series b
    ON a.marketId=b.marketId
   AND a.selectionId=b.selectionId
   AND b.id < a.id
  GROUP BY a.marketId, a.selectionId, a.id
),
diffs AS (
  SELECT
      a.marketId,
      a.selectionId,
      ABS(a.odd - b.odd)/a.odd AS rel_change
  FROM oc_series a
  JOIN prev p  ON a.id=p.id_a
  JOIN oc_series b ON b.id=p.id_prev
  WHERE a.odd IS NOT NULL AND b.odd IS NOT NULL
)
SELECT
    marketId,
    selectionId,
    ROUND(AVG(rel_change),4) AS avg_volatility,
    MIN(a.odd) AS range_low,
    MAX(a.odd) AS range_high
FROM oc_series a
LEFT JOIN diffs d USING(marketId,selectionId)
GROUP BY marketId,selectionId
