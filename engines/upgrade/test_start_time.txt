SIGNALS_DB_PATH = "/Volumes/Sonnics/Race Tool - Sandbox/analytics_beta/data/bets.db"

def save_signal_to_db(signal):
    try:
        conn = sqlite3.connect(SIGNALS_DB_PATH)
        cursor = conn.cursor()
        cursor.execute("""
            INSERT OR IGNORE INTO signals (marketId, selectionId, strategy_name, signal_type, timestamp)
            VALUES (?, ?, ?, ?, ?)
        """, (
            signal.get("marketId"),
            signal.get("selectionId"),
            signal.get("strategy_name"),
            signal.get("signal_type", signal.get("strategy_name")),
            signal.get("timestamp")
        ))
        conn.commit()
        conn.close()
        logging.info(f"📝 [DB] Signal saved to bets.db for runner {signal.get('selectionId')}")
    except Exception as e:
        logging.error(f"❌ [DB] Failed to save signal: {e}")
