    distance_band,
    fav_rank_bin,
    surface_type,
    day_of_week,
    COUNT(*)            AS bins,
    ROUND(SUM(net_pnl), 2) AS total_pnl,
    ROUND(AVG(mae_mean), 3) AS avg_mae,
    ROUND(SUM(weight_applied), 1) AS total_weight,
    ROUND(SUM(p1_alpha) / NULLIF(SUM(p1_alpha + p1_beta), 0), 3) AS p1_mean,

CREATE VIEW v_mastery_training AS
WITH
  cash AS (
    SELECT 
        marketId,
        MAX(CAST(json_extract(json_payload,'$.cashout_total') AS REAL))   AS cashout_total,
        MAX(CAST(json_extract(json_payload,'$.liability_total') AS REAL)) AS liability_total
      FROM mastery_cache
     WHERE event_type='cashout_tick'
     GROUP BY marketId
  ),
